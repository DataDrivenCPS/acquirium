\section{Scope \& objectives}

\begin{frame}{Problem \& objectives}
  Acquirium stores stream metadata in an RDF graph (Oxigraph) alongside
  TimescaleDB series. Units and quantity kinds are identified by canonical
  \textbf{QUDT} URIs, but sources and users supply free text
  (\code{"mg/L"}, \code{"kg"}, \code{"gallon per minute"}).

  \begin{block}{Objectives}
    \begin{enumerate}
      \item Resolve free text / symbols to a canonical ontology or QUDT
            URI, with ranking and optional context disambiguation.
      \item Convert numeric values between two unit URIs.
      \item Do both without mutating ingested data.
    \end{enumerate}
  \end{block}
\end{frame}

\begin{frame}{Requirements}
  \small
  \begin{tabular}{@{}p{0.13\textwidth}p{0.83\textwidth}@{}}
    \hline
    \req{R1} & Ingest does not normalize or rewrite submitted semantics:
      unit / quantity-kind values are stored as given, never canonicalized
      at ingest. (RDF set semantics still apply -- see Ingest contract.) \\
    \req{R2} & Resolution is deterministic for exact inputs; ranked and
      reproducible for fuzzy inputs. \\
    \req{R3} & Resolution and conversion are separate concerns with
      separate entry points. \\
    \req{R4} & Context can only reorder candidates, never introduce one. \\
    \req{R5} & After \code{insert\_graph} returns, the just-inserted
      concepts are resolvable: the index is refreshed synchronously,
      incrementally where possible. \\
    \req{R6} & Absence of the QUDT graph degrades gracefully, never
      hard-fails resolution. \\ \hline
  \end{tabular}
\end{frame}

    \code{TextMatch/qudt\_store.py} & QUDT vocabulary parse / cache / diff \\

    \code{TextMatch/resolver.py} & \code{ConceptResolver}: resolution

    \code{internals/qudt\_units.py} & \code{QUDTUnitConverter}: exact
        resolution $+$ conversion \\ \hline
  \end{tabular}
\end{frame}

\begin{frame}{Separation of concerns \req{R3}}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \textbf{Normalization}\\[2pt]
    text / symbol $\rightarrow$ URI
    \begin{itemize}
      \item ranked, context-aware
      \item \code{ConceptResolver}
      \item \code{GET /resolve\_text}
    \end{itemize}
    \column{0.5\textwidth}
    \textbf{Conversion}\\[2pt]
    URI $+$ value $\rightarrow$ value
    \begin{itemize}
      \item deterministic arithmetic
      \item \code{QUDTUnitConverter}
      \item \code{/resolve\_unit},
            \code{/conversion\_factors}
    \end{itemize}
  \end{columns}
  \vspace{6pt}
  \begin{block}{Shared substrate}
    A single lazily-loaded \code{QUDTUnitConverter} instance is shared.
    \code{ConceptResolver} invokes only its \emph{resolution} methods;
    conversion math is reached only via the conversion endpoints.
  \end{block}
\end{frame}

\section{Domain model}

\begin{frame}[fragile]{Data types}
  \begin{semiverbatim}\scriptsize
ResolveResult   uri, kind, label, score, matched_surface,
                match_stage in {exact, semantic}, related: tuple[uri]

concept (index) uri, kind, label, surfaces: list[str],
                symbol?, ucum?, related: list[uri]

UnitDefinition  uri, label?, symbol?, quantity_kind?,
                quantity_kinds: tuple, multiplier, offset,
                dimension_vector?

embedding_status[idx]  state in {idle,building,ready,error},
                concepts, surfaces, error?, last_built?, duration_s?
  \end{semiverbatim}
  \code{related} is the cross-reference set used by context rerank:
  for a unit, its quantity kinds; for a quantity kind, its applicable
  units.
\end{frame}

\section{Ingest \& index lifecycle}

\begin{frame}[fragile]{Ingest contract \req{R1}}
  \begin{semiverbatim}\scriptsize
POST /insert_graph -> Manager.insert_graph -> GraphStore.insert_graph:
    incoming.parse(data, fmt)
    for triple in incoming: main.add(triple)     # verbatim
    refresh_union()
  \end{semiverbatim}
  \begin{itemize}
    \item Specification: submitted triples are parsed and stored without
          semantic rewriting or unit canonicalization
          (\code{qudt:hasUnit "mg/L"} stays the literal \code{"mg/L"}).
    \item Not a byte- or document-level guarantee: RDF set semantics
          apply -- duplicate triples coalesce, blank nodes may be
          relabeled, statement order is insignificant; \code{replace=True}
          first clears the main graph; the queried union view adds the
          \code{owl:imports} closure.
    \item Rationale: normalization is a resolution-time service applied to
          queries, not to storage. \code{insert\_graph} then refreshes the
          embedding index before returning (next section).
  \end{itemize}
\end{frame}

\begin{frame}{Index construction}
  Two embedding indexes, each an \code{EmbeddingMatcher}:
  \begin{itemize}
    \item \textbf{graph index} -- concepts extracted from the union graph
          by four SPARQL queries (class, predicate, unit, quantity\_kind);
          unit/QK rows also surface \code{qudt:symbol}/\code{ucumCode}.
          Graph-defined concepts take precedence over the broad vocabulary.
    \item \textbf{QUDT index} -- the QUDT vocabulary via \code{QUDTStore}:
          HTTP-first parse with local \code{ontologies/qudt\_*.ttl}
          fallback; gzip-JSON cache diffed across restarts for incremental
          index updates.
  \end{itemize}
  Embedding model \code{bge-small-en-v1.5}; vectors L2-normalized and
  disk-cached. State is reported per index via \code{GET /embedding\_status}.
\end{frame}

\begin{frame}[fragile]{Index refresh on ingest \req{R5}}
  \begin{semiverbatim}\scriptsize
insert_graph(...):                       # synchronous, in-call
  graph_store.insert_graph(...)
  _refresh_graph_index(replace=replace):
    concepts = _extract_concepts_for_embedding()
    if replace or not matcher.is_ready:
        matcher.build_index(concepts)               # full
    else:
        added/removed = diff(concepts, get_indexed_uris())
        matcher.update_index(added, removed)         # incremental
  _notify_graph_change()
  \end{semiverbatim}
  \begin{itemize}
    \item Synchronous: when \code{insert\_graph} returns, the inserted
          concepts are resolvable. No async path, no readiness protocol.
    \item Incremental by default -- only new concepts are embedded; a full
          rebuild only on \code{replace=True} or first build.
    \item Diff is by URI (mirrors the QUDT path): a surface change on an
          \emph{existing} URI is re-embedded only on a full rebuild.
  \end{itemize}
\end{frame}

\section{Resolution specification}

\begin{frame}[fragile]{Pipeline (normative)}
  \begin{semiverbatim}\scriptsize
fetch_k = 20 if context else top_k

kind in {class, predicate}:
    R = graph.query(kind)
kind in {unit, quantity_kind}:
    G = graph.query(kind, min_score=0.8)
    if G and G[0].match_stage == exact:  R = G          # short-circuit
    else:
        D = deterministic_unit(text) if kind==unit else []
        R = combine(G + D, qudt.query(kind))
otherwise (kind is None):
    R = combine(graph.query(), qudt.query())

if context: R = rerank_by_context(R, context)
return R[:top_k]
  \end{semiverbatim}
  \code{class}/\code{predicate} resolve against graph-defined concepts
  only; QUDT participates only for \code{unit}/\code{quantity\_kind}/any.
\end{frame}

\begin{frame}[fragile]{Two-stage matcher}
  \begin{semiverbatim}\scriptsize
query(text, kind, top_k, min_score):
  1. exact   : case-sensitive surface map, then case-folded map
               -> score 1.0, match_stage = exact
  2. semantic: cosine(query_embedding, surface_vectors), kind mask,
               min_score cutoff -> match_stage = semantic
  \end{semiverbatim}
  \begin{itemize}
    \item Exact precedes semantic: QUDT symbols are case-significant
          (\code{kg} kilogram vs.\ \code{kG} kilogauss); both readings are
          retained for a later context rerank.
    \item Indexing: content hash $\rightarrow$ disk cache
          (\code{vectors.npz} $+$ \code{meta.json}); rebuild only on change.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Deterministic unit tier}
  \begin{semiverbatim}\scriptsize
deterministic_unit(text):                 # kind == "unit" only
  conv = converter_provider()             # may raise -> []
  ud = conv.resolve_unit(text)  or  conv.infer_unit(text)
  if ud.uri startswith "urn:qudt:ratio:": return []
  return [ResolveResult(ud.uri, kind=unit, score=1.0,
          match_stage=exact, related=ud.quantity_kinds)]
  \end{semiverbatim}
  \begin{itemize}
    \item Authoritative: a hit is exact (score 1.0) and carries its
          quantity kinds as \code{related}, so context still applies.
    \item Synthetic composed ratios are not emitted here; such inputs
          defer to the embedding matchers.
    \item No converter (no QUDT graph) $\Rightarrow$ empty result; the
          pipeline degrades to the matchers \req{R6}.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Merge \& context rerank}
  \begin{semiverbatim}\scriptsize
combine(primary, secondary, limit):
  stable-sort by (match_stage==exact, score) desc, primary first
  dedupe by uri; truncate to limit

rerank_by_context(R, context):
  connected = [r in R if set(context) & set(r.related)]
  rest      = [r in R if not]
  return connected + rest                 # stable within groups
  \end{semiverbatim}
  \begin{block}{Invariant \req{R4}}
    \code{rerank\_by\_context} is a stable partition of its input. It
    cannot add, remove, or invent a result; a context disjoint from every
    \code{related} set is the identity.
  \end{block}
\end{frame}

\begin{frame}{Context disambiguation (example)}
  Surface \code{"kg"} maps to two units:
  \begin{itemize}
    \item \code{unit:KiloGM}, \code{related} $\ni$ quantitykind/Mass
    \item \code{unit:KiloGAUSS}, \code{related} $\ni$
          quantitykind/MagneticFluxDensity
  \end{itemize}
  \code{resolve\_text("kg", kind=unit, context=[.../Mass])}: both surface
  as exact candidates; the rerank promotes \code{KiloGM} because its
  \code{related} set intersects the context. Context
  \code{[.../MagneticFluxDensity]} selects \code{KiloGAUSS}. Empty context
  yields the deterministic \code{combine} order.
\end{frame}

\section{Conversion specification}

\begin{frame}[fragile]{Unit resolution ladder}
  \code{QUDTUnitConverter.resolve\_unit} -- first match wins:
  \begin{semiverbatim}\scriptsize
1  direct URIRef / http(s) URI
2  UNIT[id] / UNIT[UPPER] / UNIT[norm(space,hyphen -> _)]
3  COMMON_ALIASES   (e.g. gal -> GAL_US, degc -> DEG_C)
4  literal equality over
     {rdfs:label, skos:prefLabel, qudt:symbol,
      qudt:ucumCode, qudt:uneceCommonCode}
     case-sensitive, then case-folded
   else raise UnitNotFound
  \end{semiverbatim}
  \code{infer\_unit} additionally: URI fragment / last segment; ratio
  decomposition (\code{" per "}, \code{"/"}) preferring an existing
  \code{X-PER-Y} unit, else a synthetic \code{urn:qudt:ratio:};
  label substring search.
\end{frame}

\begin{frame}[fragile]{Conversion}
  \code{UnitDefinition} carries \code{conversionMultiplier},
  \code{conversionOffset}, quantity kind(s) and dimension vector.
  \begin{block}{\code{convert(X, src, tgt)} -- \code{Decimal}, precision 28}
    \centering
    $K = (X + \text{offset}_{src})\cdot \text{mult}_{src}$
    \hspace{2em}
    $X' = K / \text{mult}_{tgt} - \text{offset}_{tgt}$
  \end{block}
  Compatibility (\code{\_check\_compatible}): dimension-vector equality
  first, then quantity-kind overlap; otherwise
  \code{IncompatibleUnits} is raised. \code{convert} resolves both
  operands via the resolution ladder before applying the formula.
\end{frame}

\section{Interface contract}

\begin{frame}[fragile]{HTTP interface}
  \scriptsize
  \begin{tabular}{@{}p{0.30\textwidth}p{0.64\textwidth}@{}}
    \hline
    \textbf{Endpoint} & \textbf{Contract} \\ \hline
    \code{GET /resolve\_text}\newline
      params: \code{text, kind?, top\_k, min\_score, context?} &
      \code{matches}: ordered \code{ResolveResult} (the pipeline). \\[4pt]
    \code{POST /resolve\_unit}\newline
      \code{\{identifier\}} &
      \code{uri, label, symbol, quantity\_kind, multiplier, offset}.
      Deterministic; \code{UnitNotFound} $\rightarrow$ 4xx. \\[4pt]
    \code{POST /conversion\_factors}\newline
      \code{\{from\_unit, to\_unit\}} &
      per-side multiplier/offset $+$ \code{compatible}; caller applies
      the formula. \\[4pt]
    \code{GET /embedding\_status} &
      per index: \code{state, concepts, surfaces, last\_built,
      duration\_s}. \\ \hline
  \end{tabular}
  \vspace{3pt}\\
  \code{insert\_graph} refreshes the graph index synchronously before
  responding; no readiness handshake is required of the caller.
\end{frame}

\section{Invariants, failure modes, verification}

\begin{frame}{Invariants \& failure modes}
  \textbf{Invariants}
  \begin{itemize}
    \item \req{R1} ingest does not rewrite semantics; \req{R4} context is
          a stable partition; \req{R5} the index reflects an ingest by the
          time \code{insert\_graph} returns.
    \item Exact-stage hits outrank semantic ones; graph-defined concepts
          outrank QUDT on ties.
  \end{itemize}
  \textbf{Failure modes \& degradation}
  \begin{itemize}
    \item No QUDT graph $\Rightarrow$ deterministic tier yields nothing;
          pipeline falls back to matchers \req{R6}.
    \item \code{UnitNotFound} / incompatible units $\Rightarrow$ explicit
          errors at the conversion endpoints, never silent coercion.
    \item Index \code{error} state is surfaced via
          \code{/embedding\_status}; resolution serves the last good index.
  \end{itemize}
\end{frame}

\begin{frame}{Verification \& operability}
  \textbf{Verification}
  \begin{itemize}
    \item \code{tests/unit/test\_concept\_resolver.py} -- pipeline policy,
          no services required.
    \item \code{tests/test\_resolve\_text.py} -- accuracy corpus, context
          cases, latency profile.
    \item \code{tests/test\_resolver\_microbench.py},
          \code{scripts/benchmark/resolver\_latency.py} -- per-tier
          performance bounds.
  \end{itemize}
  \textbf{Operability}
  \begin{itemize}
    \item \code{scripts/inspect\_resolver.py} -- one-shot / batch / REPL
          inspection against a live server; query items and field records
          (two-pass sibling context).
    \item \code{/embedding\_status} reports per-index build state for
          monitoring.
  \end{itemize}
\end{frame}

\begin{frame}{Summary}
  \begin{itemize}
    \item Ingest is faithful; normalization is a resolution-time service.
    \item A single resolution pipeline composes data-graph matching, a
          deterministic QUDT tier, vocabulary matching, and context rerank.
    \item Conversion is a separate, deterministic engine reached only
          through its own endpoints.
    \item The index is refreshed synchronously on ingest (incremental
          where possible); degradation and ranking are specified.
  \end{itemize}
\end{frame}
